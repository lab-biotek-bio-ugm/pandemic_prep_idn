# Runner environment for the lint and test jobs.
#
# Deliberately not a superset of envs/*.yaml. CI never runs blast, hyphy, mafft,
# seqkit, iqtree or augur, so those are absent by design; the test suite fakes
# their outputs instead. Only the libraries that the pipeline scripts import at
# module scope belong here, because tests/conftest.py loads several of those
# scripts during collection.
#
# Bounds rather than exact pins: a single maintainer should not have CI break on
# a stale pin, and the upper bounds are where the real risk lives. Versions
# verified working: pytest 9.1.1, pyyaml 6.0.3, biopython 1.87, pandas 2.3.3,
# numpy 2.5.1, primer3-py 2.3.0, openpyxl 3.1.5, pyflakes 3.4.0.
pytest>=8,<10
pyyaml>=6,<7
biopython>=1.83,<2

# pandas 3.0 changed enough of the DataFrame API that scripts/m10_insilico_pcr.py
# and scripts/export_supplementary.py are not known to work on it. Raise this
# bound only after running the suite against it.
pandas>=2.2,<3

numpy>=1.26,<3
primer3-py>=2.0,<3
openpyxl>=3.1,<4

# Still required by tests/test_scripts_execute.py, which shells out to
# `python -m pyflakes`. Ruff's F821 now covers the same ground in the lint job;
# keeping pyflakes installed means that test reports honestly instead of
# silently skipping through its pytest.importorskip.
pyflakes>=3.2,<4
